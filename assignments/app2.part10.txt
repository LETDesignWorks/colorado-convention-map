function renderSelectionControls() {
  const count = selectedCongregations.size;

  els.assignSelectedButton.disabled = !isAdmin() || count === 0 || !els.assignmentDestination.value;

  els.clearSelectionButton.disabled = count === 0;
  renderStats();
}
