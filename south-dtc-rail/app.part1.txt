function escapeHtml(value = '') {
  return String(value ?? '').replace(/[&<>'"]/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;'
  }[c]));
}
function clean(value = '') { return String(value ?? '').trim(); }
function parseLines(value = '') {
  return String(value).split(/[\n;]+/).map(v => v.trim()).filter(Boolean);
}
